<?php
ini_set('max_execution_time', 300);
include("../action.php");

header('Content-Type: application/json');

$sessionid = $obj->getvalfield("m_session", "sessionid", "status=1");

$inputJSON = file_get_contents('php://input');
$arr = json_decode($inputJSON, true);

if (!$arr) {
    echo json_encode(["status" => false, "message" => "Invalid JSON"]);
    exit;
}

$machine_userid = trim($arr['EmployeeID'] ?? '');
$punch          = trim($arr['PunchTime'] ?? '');
$deviceid       = trim($arr['SerialNo'] ?? ($arr['DeviceID'] ?? ''));

if ($machine_userid == '' || $punch == '' || $deviceid == '') {
    echo json_encode(["status" => false, "message" => "Required fields missing"]);
    exit;
}

$attendance_stamp = date("Y-m-d H:i:s", strtotime($punch));
$attendance_date  = date("Y-m-d", strtotime($attendance_stamp));
$attendance_time  = date("H:i:s", strtotime($attendance_stamp));
$month            = date("m", strtotime($attendance_stamp));
$year             = date("Y", strtotime($attendance_stamp));
$createdate       = date("Y-m-d");
$lastupdated      = date("Y-m-d H:i:s");

$IN_MACHINE  = '6728322120001025';
$OUT_MACHINE = '6728322120001144';

if ($deviceid == $IN_MACHINE) {
    $mode = 'IN';
} elseif ($deviceid == $OUT_MACHINE) {
    $mode = 'OUT';
} else {
    $mode = 'AUTO';
}

/* ================= EMPLOYEE FIND ================= */

$emp_unit = $obj->getvalfield(
    "att_machine_master",
    "unit_id",
    "machine_id='$deviceid'"
);

if ($emp_unit > 0) {
    $emp_id = $obj->getvalfield(
        "employee_master",
        "emp_id",
        "biomatric_id='$machine_userid' AND unit_id='$emp_unit'"
    );
} else {
    $emp_id = $obj->getvalfield(
        "employee_master",
        "emp_id",
        "biomatric_id='$machine_userid'"
    );
}

if ($emp_id <= 0) {
    echo json_encode(["status" => false, "message" => "Employee not found"]);
    exit;
}

$emp_data = $obj->select_record("employee_master", ['emp_id' => $emp_id]);

$department_id       = $emp_data['department_id'] ?? 0;
$unit_id             = $emp_data['unit_id'] ?? 0;
$office_working_hour = $emp_data['shift_id'] ?? 0;

/* ================= SHIFT FIND ================= */

$sql = "
SELECT *
FROM (
    SELECT 
        shift_id,
        in_time,
        out_time,
        is_cross_day,
        working_hour,
        min_working_hrs,

        CASE 
            WHEN is_cross_day = 0 
                 AND '$attendance_time' BETWEEN in_time AND out_time
            THEN 0

            WHEN is_cross_day = 1 
                 AND ('$attendance_time' >= in_time OR '$attendance_time' <= out_time)
            THEN 0

            ELSE 1
        END AS priority,

        ABS(TIME_TO_SEC(TIMEDIFF(in_time, '$attendance_time'))) AS time_diff

    FROM shift_master
    WHERE unit_id = '$unit_id'
    AND working_hour = '$office_working_hour'
) AS shifts
ORDER BY priority ASC, time_diff ASC
LIMIT 1
";

$res = $obj->executequery($sql);
$shift_data = $res[0] ?? [];

if (!$shift_data) {
    echo json_encode(["status" => false, "message" => "Shift not found"]);
    exit;
}

$shift_id          = $shift_data['shift_id'];
$shift_working_hr  = $shift_data['working_hour'];

$linked_attendance_id = 0;
$is_cross_day_flag    = 0;
$process_message      = "Attendance saved successfully";

/* ================= IN LOGIC ================= */

if ($mode == 'IN') {

    $duplicateCheck = $obj->getvalfield(
        "attendance_log",
        "count(*)",
        "emp_id='$emp_id'
         AND in_status='IN'
         AND TIMESTAMPDIFF(MINUTE, attendance_stamp, '$attendance_stamp') BETWEEN 0 AND 5"
    );

    if ($duplicateCheck > 0) {
        echo json_encode(["status" => true, "message" => "Duplicate IN ignored"]);
        exit;
    }

    $last_att_id = $obj->getvalfield(
        "attendance_entry",
        "attendance_id",
        "emp_id='$emp_id'
         ORDER BY attendance_id DESC
         LIMIT 1"
    );

    $canCreateNewIn = true;

   // Agar last attendance allowed range ke andar hai,
// to naya attendance_entry create mat karo.
// Sirf log insert hoga.

if ($last_att_id) {

    $last_data = $obj->select_record(
        "attendance_entry",
        ['attendance_id' => $last_att_id]
    );

    $last_in_datetime = strtotime($last_data['attendance_date'] . ' ' . $last_data['intime']);

    $last_shift_data = $obj->select_record(
        "shift_master",
        ['shift_id' => $last_data['shift_id']]
    );

    $last_working_hour = $last_shift_data['working_hour'] ?? $shift_working_hr;

    $morning_shift_start = $obj->getvalfield(
        "shift_master",
        "MIN(in_time)",
        "unit_id='$unit_id'
         AND working_hour='$last_working_hour'"
    );

    if (!$morning_shift_start) {
        $morning_shift_start = '06:00:00';
    }

    $buffer_hours = 2;

    $waitLimitTime = strtotime(
        $last_data['attendance_date'] . " " . $morning_shift_start . " +1 day +$buffer_hours hours"
    );

    if (
        strtotime($attendance_stamp) > $last_in_datetime &&
        strtotime($attendance_stamp) <= $waitLimitTime
    ) {
        $linked_attendance_id = $last_att_id;
        $process_message = "IN logged only, linked with existing attendance";
        $canCreateNewIn = false;
    }
}

    if ($canCreateNewIn) {

        $form_data = [
            'emp_id'            => $emp_id,
            'shift_id'          => $shift_id,
            'intime'            => $attendance_time,
            'attendance_date'   => $attendance_date,
            'machineid'         => $deviceid,
            'entry_type'        => 'machine',
            'in_status'         => 'IN',
            'attendance_status' => 'Incomplete',
            'month'             => $month,
            'year'              => $year,
            'unit_id'           => $unit_id
        ];

        $linked_attendance_id = $obj->insert_record("attendance_entry", $form_data);
        $process_message = "New IN created";
    }
}

/* ================= OUT LOGIC ================= */

elseif ($mode == 'OUT') {

    $outDateTime = strtotime($attendance_stamp);
    $updated     = false;

    /*
        Last attendance open ho ya closed,
        agar OUT allowed range ke andar hai,
        to same attendance_entry update hogi.
    */

    $prev_att_id = $obj->getvalfield(
        "attendance_entry",
        "attendance_id",
        "emp_id='$emp_id'
         AND attendance_date <= '$attendance_date'
         ORDER BY attendance_date DESC, attendance_id DESC
         LIMIT 1"
    );

    if ($prev_att_id) {

        $prev_data = $obj->select_record(
            "attendance_entry",
            ['attendance_id' => $prev_att_id]
        );

        $inDateTime = strtotime($prev_data['attendance_date'] . ' ' . $prev_data['intime']);

        $prev_shift_id = $prev_data['shift_id'];

        $prev_shift_data = $obj->select_record(
            "shift_master",
            ['shift_id' => $prev_shift_id]
        );

        $prev_working_hour = $prev_shift_data['working_hour'] ?? $shift_working_hr;

        $morning_shift_start = $obj->getvalfield(
            "shift_master",
            "MIN(in_time)",
            "unit_id='$unit_id'
             AND working_hour='$prev_working_hour'"
        );

        if (!$morning_shift_start) {
            $morning_shift_start = '06:00:00';
        }

        $buffer_hours = 2;

        $waitLimitTime = strtotime(
            $prev_data['attendance_date'] . " " . $morning_shift_start . " +1 day +$buffer_hours hours"
        );

        if (
            $outDateTime > $inDateTime &&
            $outDateTime <= $waitLimitTime
        ) {

            $working_seconds = $outDateTime - $inDateTime;
            $working_hours   = gmdate("H:i:s", $working_seconds);

            $update_data = [
                'outtime'           => date('H:i:s', $outDateTime),
                'working_hours'     => $working_hours,
                'attendance_status' => 'Present',
                'out_status'        => 'OUT',
                'machineid'         => $deviceid,
                'lastupdated'       => $lastupdated
            ];

            $obj->update_record(
                "attendance_entry",
                ['attendance_id' => $prev_att_id],
                $update_data
            );

            $linked_attendance_id = $prev_att_id;
            $is_cross_day_flag = ($attendance_date != $prev_data['attendance_date']) ? 1 : 0;
            $updated = true;
            $process_message = "OUT updated within allowed range";

        } else {
            $linked_attendance_id = 0;
            $process_message = "OUT logged only, out of allowed range";
        }

    } else {
        $linked_attendance_id = 0;
        $process_message = "OUT logged only, no attendance found";
    }
}

/* ================= AUTO MODE ================= */

else {
    $process_message = "AUTO mode logged only";
}

/* ================= LOG INSERT ================= */

$form_data_log = [
    'emp_id'             => $emp_id,
    'department_id'      => $department_id,
    'machine_userid'     => $machine_userid,
    'intime'             => $attendance_time,
    'attendance_stamp'   => $attendance_stamp,
    'attendance_date'    => $attendance_date,
    'createdate'         => $createdate,
    'in_status'          => $mode,
    'machineid'          => $deviceid,
    'entry_type'         => 'machine',
    'month'              => $month,
    'year'               => $year,
    'unit_id'            => $unit_id,
    'sessionid'          => $sessionid,
    'createtime'         => $attendance_time,
    'punch_pair_id'      => $linked_attendance_id,
    'ref_attendance_id'  => $is_cross_day_flag
];

$alreadyLogged = $obj->getvalfield(
    "attendance_log",
    "count(*)",
    "emp_id='$emp_id'
     AND in_status='$mode'
     AND machineid='$deviceid'
     AND attendance_stamp='$attendance_stamp'"
);

if ($alreadyLogged == 0) {
    $obj->insert_record("attendance_log", $form_data_log);
}

echo json_encode([
    "status"        => true,
    "message"       => $process_message,
    "mode"          => $mode,
    "attendance_id" => $linked_attendance_id
]);
?>
